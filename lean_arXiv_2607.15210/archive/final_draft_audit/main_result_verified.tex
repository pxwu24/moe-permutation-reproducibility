% Suggested abstract correction:
% In this work, we prove that for every $p>0$ there exists a pair of
% finite-dimensional quantum channels whose minimum output $p$-R\'enyi
% entropies violate additivity.
% The dimensions may depend on p; the proof does not establish one pair
% that works simultaneously for all positive orders.

\begin{proof}[Proof of Theorem~\ref{thm:main}]
Fix \(p>0\), and take \(r=5\), \(\gamma=375\), and \(t=1/376\).
Taylor's formula, with its continuous interpretation at \(p=1\), gives
\begin{equation}
    \frac{B_{p,5}(375)}{375p}
    =\frac1{15}\int_1^{16}(16-s)s^{p-2}\,ds
    \geq1-\frac{\log16}{15}>\frac45.
\end{equation}
Indeed, \(s^p\geq1\) on \([1,16]\), and \(\log16<3\).
Thus \(\Delta_p:=B_{p,5}(375)-300p>0\).
Theorem~\ref{thm:entropy-asymptotics-antisymmetric} yields, almost surely,
\begin{equation}
    \lim_{n\to\infty}\left[
      2S_p^{\min}(\Theta_n)
      -S_p\!\left((\Theta_n\otimes\overline\Theta_n)(\psi_n^+)\right)
    \right]
    =\frac{\Delta_p}{k^2}+o(k^{-2}).
\end{equation}
Choose \(k\geq20\) sufficiently large that this limit is positive,
and then choose \(n\) sufficiently large. Since complex conjugation
preserves minimum output entropy,
\begin{equation}
    S_p^{\min}(\Theta_n\otimes\overline\Theta_n)
    \leq S_p\!\left((\Theta_n\otimes\overline\Theta_n)(\psi_n^+)\right)
    <S_p^{\min}(\Theta_n)+S_p^{\min}(\overline\Theta_n).
\end{equation}
This proves the assertion for the fixed order \(p\).
\end{proof}
