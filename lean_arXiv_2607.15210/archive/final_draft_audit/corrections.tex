% 2. A short, explicit proof of full local support.
% Replace the proof of Lemma \ref{lem:haar-projection-partial-trace}.
\begin{proof}
Let $G\in\mb M_{nk,d}$ have independent standard complex Gaussian entries,
and write $G=(G_1^T,\ldots,G_k^T)^T$ with $G_i\in\mb M_{n,d}$,
after ordering the tensor basis by the $B$ coordinate.
Almost surely, $G$ has full column rank and
$P=G(G^*G)^{-1}G^*$ is a Haar projection. Put $W=(G^*G)^{-1}>0$.
Then
\begin{equation}
 \Tr_BP=\sum_{i=1}^kG_iWG_i^*,\qquad
 \ker(\Tr_BP)=\bigcap_{i=1}^k\ker G_i^*
             =\ker [G_1\ \cdots\ G_k]^*.
\end{equation}
The concatenated matrix has independent Gaussian entries, so its rank
is $\min\{n,kd\}$ almost surely. This proves both assertions.
\end{proof}

% 3. Opening of the random-compression lemma: fix the sequence notation.
Fix an integer $k\geq1$. For each $n$, let $P_n$ be a Haar-distributed
projection of rank $d_n$ on $\mb C^n\otimes\mb C^k$, where
$1\leq d_n\leq nk$ and $d_n/(kn)\to t\in(0,1)$.
In the standard tensor product basis, write
$P_n=\sum_{i,j=1}^kP_{ij}^{(n)}\otimes|i\rangle\langle j|$.

% 4. In its proof, distinguish the two Choi conventions.
In our input-first convention, $J_{\varphi_a}=A\otimes I_k$.
The block-modification theorem uses the output-first convention
$C_{\varphi_a}=I_k\otimes A$. Its full spectral projections have the
form $I_k\otimes\Pi_\lambda$, whose partial trace over the second
factor is $\rank(\Pi_\lambda)I_k$. Thus the required unitarity
condition holds.

% 7. Optional clarification of the negative inverse-Cauchy branch,
% inserted after the log-potential identity in the Bell appendix.
For $a_i>0$, put $\Phi_X(w)=1+k\sum_i g_t(a_iw/k)$.
This function is strictly increasing on $(-\infty,0)$, with limits
$1-k^2t<0$ and $1$. It therefore has a unique zero $w_X<0$.
On $(w_X,0)$, $K_X(w)=\Phi_X(w)/w$ is strictly decreasing from $0$
to $-\infty$. The inverse-Cauchy germ consequently continues along
this interval, which justifies the evaluation at $z=0$.
