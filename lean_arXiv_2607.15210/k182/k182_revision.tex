\subsection{Improved output dimension threshold}

For $p>0$, define
\begin{equation}
 k_{\mathrm{high}}(p)
 :=\min\left\{k\geq2:\ \exists t\in(k^{-2},1),\quad
 S_p(\lambda^{\mathrm{Bell}}_{k,t})
 <2\min_{\sigma\in\mathscr K_{k,t}}S_p(\sigma)\right\},
 \label{eq:k-high-def}
\end{equation}
with $\min\varnothing=+\infty$. This threshold concerns the present
ensemble and Bell-state witness.

\begin{proposition}[Certified output dimension]
\label{prop:certified-k182}
For $k=182$ and $t=27/100000$,
\begin{equation}
 2\min_{\sigma\in\mathscr K_{182,t}}S_1(\sigma)
 -S_1(\lambda^{\mathrm{Bell}}_{182,t})
 >\frac{477}{10^6}.
 \label{eq:certified-k182-gap}
\end{equation}
In particular, $k_{\mathrm{high}}(1)\leq182$.
\end{proposition}

\begin{proof}
Put $L=162513/10^6$ and $z=2077/2000$. The assumptions of
Lemma~\ref{lem:finite-k-entropy-reduction} hold, and the scalar bound in
Lemma~\ref{lem:finite-k-scalar-certificate} satisfies
\begin{equation}
 -1.660\times10^{-8}
 <\mathcal U_{182,t}(L,z)<-1.659\times10^{-8}.
\end{equation}
Thus every $\lambda\in\Lambda_{182,t}$ has $\lambda_i\leq L$, and
\begin{equation}
 \min_{\sigma\in\mathscr K_{182,t}}S_1(\sigma)
 \geq h_{182}(L),\qquad
 h_k(L):=-L\log L-(1-L)\log\frac{1-L}{k-1}.
\end{equation}
The accompanying Python certificate uses outward-rounded interval
arithmetic to verify
\begin{align}
 4.79748729461169
 &<h_{182}(L)<4.79748729461170,\\
 9.59449702812108
 &<S_1(\lambda^{\mathrm{Bell}}_{182,t})<9.59449702812109.
\end{align}
These bounds imply \eqref{eq:certified-k182-gap}.
\end{proof}

Together with the output-set and Bell-output limits, this gives
finite-dimensional channel pairs with output dimension $182$ that
violate additivity of minimum output von Neumann entropy. For comparison,
the random-Stinespring/Bell-state criterion of Belinschi, Collins, and
Nechita first detects a violation at $k=183$~\cite{Belinschi_2016}.
We do not claim that $182$ is minimal: equality in
$k_{\mathrm{high}}(1)\leq182$ would also require excluding every
$2\leq k\leq181$ and every admissible $t$.

\subsection{Certification of the finite-dimensional entropy bound}
\label{app:k182-certificate}

% Mathematical reduction underlying the finite-dimensional certificate.
\begin{lemma}[Reduction of the entropy minimum]
\label{lem:finite-k-entropy-reduction}
Let $k\geq2$, and suppose that
\begin{equation}
 \frac1{k(k-1)}<t<1-\frac1k,
 \qquad
 U_{k,t}:=\left(\sqrt{t(1-k^{-1})}
                  +\sqrt{(1-t)k^{-1}}\right)^2<\frac14.
\end{equation}
The minimum of $S_1$ over $\mathscr K_{k,t}$ is attained at a
spectrum of the form
\begin{equation}
 \left(x,\frac{1-x}{k-1},\ldots,\frac{1-x}{k-1}\right),
 \qquad x>\frac1k.
\end{equation}
Consequently, if every $\lambda\in\Lambda_{k,t}$ satisfies
$\lambda_i\leq L<1$, where $L\geq1/k$, then
\begin{equation}\label{eq:finite-k-entropy-bound}
 \min_{\sigma\in\mathscr K_{k,t}}S_1(\sigma)
 \geq-L\log L-(1-L)\log\frac{1-L}{k-1}.
\end{equation}
\end{lemma}

\begin{proof}
Put $C(u)=\sum_i c_t(u_i)$, $s=\sum_i u_i$, and
$F(u)=S_1(u/s)$. The assumptions give $s>0$ and $u_i\leq U_{k,t}<1/4$
on $\mathscr D_{k,t}$, and exclude vectors with only one positive
coordinate. A minimizer exists by compactness. Its value is strictly
less than $\log k$, since a small nonconstant perturbation of
$(t,\ldots,t)$ is feasible.

We first exclude zero coordinates. If two positive coordinates are
unequal, transfer $\varepsilon$ from the smaller to the larger and
replace a zero by $M\varepsilon^2$. The transfer decreases $F$ by a
positive multiple of $\varepsilon$, whereas filling the zero changes
$F$ by $O(\varepsilon^2|\log\varepsilon|)$. Since
\begin{equation}
 c_t(v)=t-2\sqrt{t(1-t)}\sqrt v+O_t(v),\qquad v\downarrow0,
\end{equation}
choosing $M$ sufficiently large makes $C$ decrease. If all positive
coordinates are equal, transfer $\varepsilon$ between two of them
and replace a zero by $M\varepsilon^4$. The transfer now decreases
$F$ by a positive multiple of $\varepsilon^2$, while the same
argument makes $C$ decrease and the entropy cost of filling the zero
is $O(\varepsilon^4|\log\varepsilon|)$. Both cases contradict
minimality.

Thus the minimizer has positive coordinates. The constraint is
active, since otherwise a transfer between unequal coordinates
would decrease $F$. Since $\nabla C\neq0$ on the constraint boundary, the multiplier rule
gives $\mu>0$ such that
\begin{equation}
 f(u_i)=\frac{\sum_j u_j\log u_j}{s},
 \qquad f(v):=\log v-\mu c_t'(v).
\end{equation}
Indeed, a zero multiplier would force all coordinates to be equal.
Now
\begin{equation}
 c_t''(v)=\frac{\sqrt{t(1-t)}}{2[v(1-v)]^{3/2}},
 \qquad
 f'(v)=\frac1v-\mu c_t''(v).
\end{equation}
Since $\sqrt v(1-v)^{3/2}$ is strictly increasing on $(0,1/4)$,
$f'$ changes sign at most once, from negative to positive.
Hence the coordinates take exactly two values $a<b$, with $f'(b)>0$.
If $b$ occurred twice, the tangent direction transferring mass
between these two coordinates would give the constrained second
variation
\begin{equation}
 \frac2s\left(\mu c_t''(b)-\frac1b\right)
 =-\frac2s f'(b)<0,
\end{equation}
contradicting minimality. Therefore $b$ occurs once. The final claim
follows because the entropy of the displayed spectrum decreases
for $x\geq1/k$.
\end{proof}

\begin{lemma}[A scalar upper certificate for the largest eigenvalue]
\label{lem:finite-k-scalar-certificate}
Assume $k\geq2$ and $t\in(k^{-2},1)$. For $0<L<1$ and $z\geq0$, put
\begin{align}
 q_t(a,z)&:=\frac{a-z+\sqrt{a^2-2az(1-2t)+z^2}}2,\\
 \mathcal U_{k,t}(L,z)
 &:=\frac zk+q_t(1-L,z)+(k-1)q_t(-L,z).
\end{align}
If $\mathcal U_{k,t}(L,z)\leq0$, then every
$\lambda\in\Lambda_{k,t}$ satisfies $\lambda_i\leq L$.
\end{lemma}

\begin{proof}
For every real $a$,
\begin{equation}
 \max_{0\leq v\leq1}\{av-zc_t(v)\}
 =q_t(a,z).
\end{equation}
This follows by maximizing a linear functional over the upper unit
semicircle after writing $v=(1+x)/2$. Apply the identity with
$a_1=1-L$ and $a_2=\cdots=a_k=-L$. For $u\in\mathscr D_{k,t}$,
\begin{equation}
 u_1-L\sum_i u_i
 \leq\frac zk+\sum_i\max_{0\leq v\leq1}\{a_iv-zc_t(v)\}
 =\mathcal U_{k,t}(L,z)\leq0.
\end{equation}
Divide by $\sum_i u_i>0$ and use permutation invariance.
\end{proof}
